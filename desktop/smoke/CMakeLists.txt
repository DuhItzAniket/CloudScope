cmake_minimum_required(VERSION 3.21)
project(cloudscope_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(OpenCV REQUIRED)
if(NOT DEFINED ORT_ROOT)
  if(DEFINED ENV{ORT_ROOT})
    set(ORT_ROOT $ENV{ORT_ROOT})
  else()
    set(ORT_ROOT "C:/tools/onnxruntime/onnxruntime-win-x64-gpu-1.22.0")
  endif()
endif()

add_executable(smoke main.cpp)
target_include_directories(smoke PRIVATE "${ORT_ROOT}/include")
target_link_directories(smoke PRIVATE "${ORT_ROOT}/lib")
target_link_libraries(smoke PRIVATE Qt6::Core ${OpenCV_LIBS} onnxruntime)
